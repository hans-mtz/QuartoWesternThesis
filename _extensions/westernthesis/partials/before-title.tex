% ---------------------------------------------------------------------------
% Western thesis preamble
% Rules: SGPS "Procedure for Thesis Formats and Content" + SGPS Word template
% (etd_template2026.docx) for the visual style.
% ---------------------------------------------------------------------------
\usepackage{setspace}
\usepackage{fancyhdr}
\usepackage{etoolbox}
\usepackage{titlesec}
\usepackage[titles]{tocloft}

% Keep all text inside the margins: no character protrusion into the margin.
\makeatletter
\@ifpackageloaded{microtype}{\microtypesetup{protrusion=false}}{}
\makeatother
\setlength{\emergencystretch}{3em}

% Body text (Word "Normal"): 12pt, 1.5 spacing (§6.3), block paragraphs with
% 12pt before each paragraph and no first-line indent.
\AtBeginDocument{\$line-spacing$spacing}
\setlength{\parindent}{0pt}
\setlength{\parskip}{12pt plus 2pt minus 1pt}

% Headings (Word "Heading 1-3"): Arial-like sans, bold; 16/16/14pt.
% (\setstretch before \fontsize: setspace's \singlespacing would reset the size.)
\newcommand{\westernheadingfont}{\normalfont\sffamily\bfseries}
\titleformat{\chapter}[display]
  {\westernheadingfont\setstretch{1}\fontsize{16}{19}\selectfont}
  {\chaptertitlename\ \thechapter}{6pt}{}
\titleformat{name=\chapter,numberless}
  {\westernheadingfont\setstretch{1}\fontsize{16}{19}\selectfont\centering}
  {}{0pt}{}
\titlespacing*{\chapter}{0pt}{-26pt}{12pt}
\titleformat{\section}
  {\westernheadingfont\setstretch{1}\fontsize{16}{19}\selectfont}{\thesection}{1em}{}
\titleformat{\subsection}
  {\westernheadingfont\setstretch{1}\fontsize{14}{17}\selectfont}{\thesubsection}{1em}{}
\titleformat{\subsubsection}
  {\westernheadingfont\setstretch{1}\fontsize{12}{15}\selectfont}{\thesubsubsection}{1em}{}
\titlespacing*{\section}{0pt}{12pt}{6pt}
\titlespacing*{\subsection}{0pt}{12pt}{6pt}
\titlespacing*{\subsubsection}{0pt}{12pt}{6pt}

% "Keywords" sub-heading at the end of the abstract (Word template: 16pt).
\newcommand{\westernkeywordsheading}{%
  \par{\westernheadingfont\fontsize{16}{19}\selectfont Keywords\par}}

% Captions (Word "Caption"): bold, "Table 1.1: ..." style.
\makeatletter
\@ifpackageloaded{caption}{}{\usepackage{caption}}
\makeatother
\captionsetup{labelfont=bf,textfont=bf,labelsep=colon}

% Footnotes (Word "Footnote Text"): 10pt (>= 9pt, §6.2); setspace keeps them
% single-spaced.
\renewcommand{\footnotesize}{\fontsize{10}{12}\selectfont}

% Bibliography (Word "biblio"): single-spaced (§6.3). The hanging indent comes
% from the CSL style.
\AtBeginEnvironment{CSLReferences}{\singlespacing}

% Table of contents and lists: entries single-spaced, as in the Word template.
\renewcommand{\cftchapfont}{\normalfont}
\renewcommand{\cftchappagefont}{\normalfont}
\renewcommand{\cftchapleader}{\cftdotfill{\cftdotsep}}
\setlength{\cftbeforechapskip}{6pt}

% List of Appendices (§1.4), filled by westernthesis.lua; printed by
% {{< western-lists >}} only when the thesis has lettered appendices.
\newlistof{appendixlist}{loa}{List of Appendices}
$if(has-appendices)$
\newcommand{\westernlistofappendices}{%
  \clearpage\phantomsection
  \addcontentsline{toc}{chapter}{List of Appendices}%
  \listofappendixlist}
$else$
\newcommand{\westernlistofappendices}{}
$endif$

% Page styles (§6.6)
%   front matter: lowercase roman, centred at the bottom
%   body:         arabic, upper right corner (chapter opening pages included)
\fancypagestyle{westernfront}{%
  \fancyhf{}%
  \fancyfoot[C]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}
\fancypagestyle{westernbody}{%
  \fancyhf{}%
$if(running-headers)$
  \fancyhead[L]{\small\nouppercase{\leftmark}}%
$endif$
  \fancyhead[R]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
  \renewcommand{\chaptermark}[1]{\markboth{\thechapter\quad ##1}{}}%
}
\fancypagestyle{westernbodyplain}{%
  \fancyhf{}%
  \fancyhead[R]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}

% Front matter: \chapter* pages use `plain`, so point `plain` at the front style.
\newcommand{\westernfrontmatter}{%
  \pagenumbering{roman}%
  \fancypagestyle{plain}[westernfront]{}%
  \pagestyle{westernfront}%
}
% Body: page 1 is the first page of Chapter 1 (inserted by westernthesis.lua).
\newcommand{\westernmainmatter}{%
  \clearpage
  \pagenumbering{arabic}%
  \fancypagestyle{plain}[westernbodyplain]{}%
  \pagestyle{westernbody}%
}
